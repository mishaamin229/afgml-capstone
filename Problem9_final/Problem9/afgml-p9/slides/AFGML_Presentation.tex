\begin{frame}{Research question and decision context}
\begin{itemize}
\item Context: bank outreach scored by a Bayesian network under a compute limit; affected: customers, outreach teams, serving engineers, auditors.
\item $Y=1$: recorded term-deposit subscription; $D=1$: selected for \emph{hypothetical} pre-contact outreach (capacity \resCapacity).
\item Question: at a fixed operation budget, does fairness-aware fallback selection give smaller $\Delta_{\mathrm{EOpp}}$ than a global fallback, utility-only selection and post-budget threshold adjustment?
\item Hypothesis: yes, at $\le0.01$ AP loss; refuted if a Holm-adjusted paired 95\% interval contains $0$ or is positive.
\end{itemize}
\end{frame}

\begin{frame}{Closest work and proposed contribution}
\begin{itemize}
\item Established: compute reductions can create group disparities (Das et al., ICML 2024; Hooker et al. 2020) --- neural models, accuracy parity.
\item Open issue: no verified study found for budget-triggered \emph{approximate PGM inference} and decision fairness (search to be extended).
\item Contribution: operation-count budgets with evidence-dependent trips; simulation isolating group-dependent evidence availability (with control); fairness-aware fallback selection.
\item Evidence: comparison with global fallback, utility-only selection, Hardt-style post-processing; ablations.
\end{itemize}
\end{frame}

\begin{frame}{Data and evaluation population}
\begin{itemize}
\item UCI Bank Marketing (dataset 222), \texttt{bank-additional-full.csv}; $Y=1$ iff \texttt{y = yes}; pre-contact variables only (\textbf{no \texttt{duration}}).
\item Protected: age band ($<30$, $30$--$59$, $\ge60$) $\times$ marital status (audit only).
\item Split: positional 60/20/20 (date-ordered release); test size \resNtest; no customer ids.
\item Model inputs: age band, job, education, default, housing, loan, pdays bin, previous bin, poutcome; \texttt{unknown} $\Rightarrow$ unobserved.
\item Groups with $<30$ relevant observations: exploratory (see backup).
\end{itemize}
\end{frame}

\begin{frame}{Fairness criterion and proposed method}
\begin{itemize}
\item Primary: $\Delta_{\mathrm{EOpp}}=\max_g t_g-\min_g t_g$, $t_g=P(D{=}1\mid Y{=}1,G{=}g)$; a missed responder is the harm. Diagnostics: $\Delta_{\mathrm{DP}}$, $\Delta_{\mathrm{EOdds}}$, fallback-rate gap, decision flips vs exact, calibration.
\item Utility: AP; allowed loss $0.01$ (validation-selected).
\item Change: if exact elimination would exceed budget $B$ ops $\Rightarrow$ table lookup. Proposed pick: $f^\star=\arg\min_f \max_g \mathrm{flip}_g(f)$ s.t.\ $\mathrm{AP\ loss}(f)\le0.01$.
\item Protected attribute at prediction time: age band is a model variable (yes); routing is group-unaware; marital status unused by models.
\end{itemize}
\end{frame}

\begin{frame}{Experimental protocol}
\begin{itemize}
\item Baselines: exact; global fallback; threshold adjustment after budgeted inference; fair-adjusted exact; LR; GBT.
\item Same variables, splits, tuning; thresholds/tables/selection from train+validation; test frozen.
\item Budgets: quantiles of natural validation op counts (primary 75\%: \resBudget\ ops); wall-clock separate.
\item Runs: \resSeeds\ seeds; paired block-bootstrap (\resNboot\ reps, date blocks); Holm over 3 comparisons.
\item Mechanism: synthetic networks (width $\times$ availability, with control); ablations; robustness.
\end{itemize}
\end{frame}
